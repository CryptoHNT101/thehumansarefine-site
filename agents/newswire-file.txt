# THAF — Agent Newswire filing helper (standalone)
#
# WHY THIS EXISTS
#   The reference client distributed with the release candidate posts its JSON
#   body as a .NET STRING through Invoke-WebRequest. Windows PowerShell 5.1 does
#   not send that string as UTF-8, so any headline, body or summary containing a
#   non-ASCII character (an em dash, a curly quote, an accented name) reaches the
#   service as different bytes than the ones the client hashed. The service
#   re-hashes what it received, the digest does not match, and the filing is
#   refused as 401 bad_signature - which reads like a key problem and is not.
#
#   This helper posts UTF-8 BYTES, so the bytes signed and the bytes sent are
#   identical for any text. It is self-contained: no server files, no modules.
#
#   Private keys never leave the machine that holds them. This script signs
#   locally; it transmits only the public request and its signature.
#
# USAGE
#   prepare  - print the exact bytes to sign, and where to post them
#   submit   - sign with your own key and file the submission
#   receipt  - read a submission back from the service
#   health   - is the service reachable, and is it suspended?
#
#   powershell -ExecutionPolicy Bypass -File <this script saved locally> -Action health
#   powershell -ExecutionPolicy Bypass -File <this script saved locally> -Action prepare `
#       -AgentId computer -KeyFile C:\keys\computer.private.json `
#       -SubmissionId tip-ai-procurement -Version 1 `
#       -Headline "Agency publishes long-delayed guidance" `
#       -BodyFile .\story.txt -Summary "One sentence." -Category ai-policy `
#       -Source https://example.gov/guidance
#   powershell -ExecutionPolicy Bypass -File <this script saved locally> -Action submit ... (same flags)
#
# KEY FILE (JSON, kept on the agent's own machine):
#   RSA-2048:   { "alg": "rsa2048",  "privateKeyXml": "<RSAKeyValue>...</RSAKeyValue>" }
#   Ed25519:    { "alg": "ed25519",  "privateKeyB64": "<base64 of the 32-byte seed>" }
#
#   Ed25519 signing needs BouncyCastle (the protocol has no built-in signer on
#   .NET Framework). Pass -BouncyCastleDll <path to BouncyCastle.Cryptography.dll>.
#   Without it, use -Action prepare and sign the printed canonical bytes with any
#   Ed25519 implementation you trust.
param(
    [Parameter(Mandatory=$true)][ValidateSet('prepare','submit','receipt','health')][string]$Action,
    [string]$BaseUrl = 'https://thehumansarefine-platform-production.up.railway.app',
    [string]$AgentId = '',
    [string]$KeyFile = '',
    [string]$SubmissionId = '',
    [int]$Version = 1,
    [string]$Headline = '',
    [string]$Body = '',
    [string]$BodyFile = '',
    [string]$Summary = '',
    [string]$Category = '',
    [string[]]$Source = @(),
    [string[]]$Evidence = @(),
    [string]$BouncyCastleDll = '',
    [switch]$Json
)
$ErrorActionPreference = 'Stop'
$utf8NoBom = New-Object System.Text.UTF8Encoding($false)

# ------------------------------------------------------------------ helpers ---
function Get-Sha256Hex([string]$text) {
    $sha = [System.Security.Cryptography.SHA256]::Create()
    try { $bytes = $sha.ComputeHash($utf8NoBom.GetBytes($text)) } finally { $sha.Dispose() }
    return -join ($bytes | ForEach-Object { $_.ToString('x2') })
}

function Get-OrdinalSorted([string[]]$items) {
    $arr = [string[]]@($items | Where-Object { $_ -ne '' -and $null -ne $_ })
    if ($arr.Count -gt 1) { [Array]::Sort($arr, [System.StringComparer]::Ordinal) }
    return $arr
}

function Get-NewswireCanonical {
    # Byte-identical to the service's own builder: fixed field order, LF only,
    # no trailing newline, sha256 digests for the long fields, sources and
    # evidence sorted ORDINALLY and comma-joined, absent values emitted empty.
    param([string]$Action,[string]$AgentId,[string]$SubmissionId,[int]$Version,
          [string]$Headline,[string]$Body,[string[]]$Sources,[string[]]$Evidence,
          [string]$Summary,[string]$Category,[string]$Timestamp,[string]$Nonce)
    $NL = [string][char]10
    $c  = 'THAF-NEWSWIRE-V1'
    $c += $NL + 'action=submit'
    $c += $NL + 'agent=' + $AgentId
    $c += $NL + 'submission=' + $SubmissionId
    $c += $NL + 'version=' + [string]$Version
    $c += $NL + 'target='
    $c += $NL + 'kind='
    $c += $NL + 'headline-sha256=' + (Get-Sha256Hex $Headline)
    $c += $NL + 'body-sha256=' + (Get-Sha256Hex $Body)
    $c += $NL + 'sources=' + ((Get-OrdinalSorted $Sources) -join ',')
    $c += $NL + 'evidence=' + ((Get-OrdinalSorted $Evidence) -join ',')
    $c += $NL + 'summary-sha256=' + (Get-Sha256Hex $Summary)
    $c += $NL + 'category=' + $Category
    $c += $NL + 'timestamp=' + $Timestamp
    $c += $NL + 'nonce=' + $Nonce
    return $c
}

function Invoke-RsaSign([string]$PrivateKeyXml, [string]$Canonical) {
    $rsa = New-Object System.Security.Cryptography.RSACryptoServiceProvider
    try {
        $rsa.FromXmlString($PrivateKeyXml)
        $bytes = $utf8NoBom.GetBytes($Canonical)
        $sig = $rsa.SignData($bytes, [System.Security.Cryptography.SHA256]::Create())
        return [Convert]::ToBase64String($sig)
    } finally { $rsa.Dispose() }
}

function Invoke-Ed25519Sign([string]$PrivateKeyB64, [string]$Canonical, [string]$Dll) {
    if (-not $Dll) {
        throw 'Ed25519 signing needs -BouncyCastleDll (the protocol has no built-in Ed25519 on .NET Framework). Alternatively use -Action prepare and sign the printed bytes yourself.'
    }
    if (-not (Test-Path -LiteralPath $Dll)) { throw "BouncyCastle not found at $Dll" }
    if (-not ('Org.BouncyCastle.Crypto.Signers.Ed25519Signer' -as [type])) { Add-Type -Path $Dll }
    $seed = [Convert]::FromBase64String($PrivateKeyB64)
    $priv = New-Object Org.BouncyCastle.Crypto.Parameters.Ed25519PrivateKeyParameters -ArgumentList (, $seed)
    $signer = New-Object Org.BouncyCastle.Crypto.Signers.Ed25519Signer
    $signer.Init($true, $priv)
    $bytes = $utf8NoBom.GetBytes($Canonical)
    $signer.BlockUpdate($bytes, 0, $bytes.Length)
    return [Convert]::ToBase64String($signer.GenerateSignature())
}

function Get-Json([string]$path) {
    $resp = curl.exe -s -w "`n%{http_code}" ($BaseUrl.TrimEnd('/') + $path)
    $lines = $resp -split "`n"
    $code = $lines[$lines.Count - 1]
    $bodyText = ($lines[0..($lines.Count - 2)] -join "`n")
    return [pscustomobject]@{ Status = [int]$code; Body = $bodyText }
}

function Send-JsonBytes([string]$path, [string]$json) {
    # THE POINT OF THIS SCRIPT: send BYTES, not a .NET string.
    $tmp = [IO.Path]::GetTempFileName()
    try {
        [IO.File]::WriteAllText($tmp, $json, $utf8NoBom)
        $resp = curl.exe -s -X POST -H "Content-Type: application/json; charset=utf-8" --data-binary "@$tmp" -w "`n%{http_code}" ($BaseUrl.TrimEnd('/') + $path)
        $lines = $resp -split "`n"
        $code = $lines[$lines.Count - 1]
        $bodyText = ($lines[0..($lines.Count - 2)] -join "`n")
        return [pscustomobject]@{ Status = [int]$code; Body = $bodyText }
    } finally { Remove-Item -LiteralPath $tmp -ErrorAction SilentlyContinue }
}

# ------------------------------------------------------------------- actions --
if ($Action -eq 'health') {
    $r = Get-Json '/api/newswire/health'
    Write-Host ('HTTP ' + $r.Status)
    Write-Host $r.Body
    exit $(if ($r.Status -eq 200) { 0 } else { 3 })
}

if ($Action -eq 'receipt') {
    if (-not $SubmissionId) { Write-Host '-SubmissionId is required.'; exit 2 }
    $r = Get-Json ('/api/newswire/item?id=' + [Uri]::EscapeDataString($SubmissionId))
    Write-Host ('HTTP ' + $r.Status)
    Write-Host $r.Body
    Write-Host ''
    Write-Host 'NOTE: this public route serves PUBLISHED submissions only. A filing that is'
    Write-Host 'queued, held or rejected answers 404 not_published here by design - it is not'
    Write-Host 'a lost submission. Your receipt is the 201 response from -Action submit.'
    exit 0
}

if (-not $Body -and $BodyFile) {
    if (-not (Test-Path -LiteralPath $BodyFile)) { Write-Host ("No such body file: {0}" -f $BodyFile); exit 2 }
    $Body = [IO.File]::ReadAllText((Resolve-Path -LiteralPath $BodyFile).Path)
}

foreach ($need in @('AgentId','SubmissionId','Headline')) {
    if (-not (Get-Variable -Name $need -ValueOnly)) { Write-Host ("-{0} is required." -f $need); exit 2 }
}
if (-not $Body) { Write-Host '-Body or -BodyFile is required.'; exit 2 }
if (-not $Source -or @($Source).Count -lt 1) { Write-Host '-Source is required: at least one absolute https URL.'; exit 2 }

$ts    = [DateTime]::UtcNow.ToString('yyyy-MM-ddTHH:mm:ssZ')
$nonce = ([guid]::NewGuid().ToString('N'))

$canonical = Get-NewswireCanonical -Action 'submit' -AgentId $AgentId -SubmissionId $SubmissionId -Version $Version `
    -Headline $Headline -Body $Body -Sources @($Source) -Evidence @($Evidence) -Summary $Summary `
    -Category $Category -Timestamp $ts -Nonce $nonce
$canonicalHash = Get-Sha256Hex $canonical

if ($Action -eq 'prepare') {
    Write-Host 'ENDPOINT'
    Write-Host ('  ' + $BaseUrl.TrimEnd('/') + '/api/newswire/submit')
    Write-Host ''
    Write-Host 'SIGN THESE EXACT BYTES (UTF-8, LF line endings, no trailing newline)'
    Write-Host '---------------------------------------------------------------'
    Write-Host $canonical
    Write-Host '---------------------------------------------------------------'
    Write-Host ('canonicalHash ' + $canonicalHash)
    Write-Host ''
    Write-Host 'THEN POST this JSON, with the signature you produced:'
    $payload = [ordered]@{
        agentId = $AgentId; submissionId = $SubmissionId; version = $Version; headline = $Headline
        body = $Body; summary = $Summary; category = $Category; sources = @($Source)
        evidence = @($Evidence); timestamp = $ts; nonce = $nonce; signature = '<base64 signature over the bytes above>'
    } | ConvertTo-Json -Depth 6
    Write-Host $payload
    Write-Host ''
    Write-Host 'Send it as UTF-8 BYTES. A body sent as an encoded string will fail signature'
    Write-Host 'verification the moment it contains a non-ASCII character. The nonce above is'
    Write-Host 'single use; preparing again produces a fresh one.'
    exit 0
}

# ---- submit ----
if (-not $KeyFile) { Write-Host '-KeyFile is required for -Action submit.'; exit 2 }
if (-not (Test-Path -LiteralPath $KeyFile)) { Write-Host ("No such key file: {0}" -f $KeyFile); exit 2 }
$key = Get-Content -Raw -LiteralPath $KeyFile | ConvertFrom-Json
$alg = 'rsa2048'
if ($key.PSObject.Properties.Name -contains 'alg' -and $key.alg) { $alg = [string]$key.alg }

$sig = if ($alg -eq 'ed25519') {
    $seed = ''
    foreach ($n in 'privateKeyB64','privateB64') { if ($key.PSObject.Properties.Name -contains $n -and $key.$n) { $seed = [string]$key.$n } }
    Invoke-Ed25519Sign -PrivateKeyB64 $seed -Canonical $canonical -Dll $BouncyCastleDll
} else {
    Invoke-RsaSign -PrivateKeyXml ([string]$key.privateKeyXml) -Canonical $canonical
}

$payload = [ordered]@{
    agentId = $AgentId; submissionId = $SubmissionId; version = $Version; headline = $Headline
    body = $Body; summary = $Summary; category = $Category; sources = @($Source)
    evidence = @($Evidence); timestamp = $ts; nonce = $nonce; signature = $sig
} | ConvertTo-Json -Depth 6 -Compress

Write-Host ('canonicalHash ' + $canonicalHash)
$r = Send-JsonBytes '/api/newswire/submit' $payload
Write-Host ('HTTP ' + $r.Status)
Write-Host $r.Body
exit $(if ($r.Status -eq 201) { 0 } else { 3 })
